# Máquina para reconocer la palabra "if" delimitada (patrón: _if_)
# Esta máquina opera sobre un alfabeto que contiene (al menos) las letras minúsculas y el espacio.
#
# Estados:
#   q1: Estado base o “listo” para comenzar a procesar una palabra.
#        – En este estado, al recibir un espacio, se permanece en q1.
#        – Si se recibe "i", se transita a q2.
#        – Cualquier otro carácter lleva al estado de rechazo (qd).
#
#   q2: Se ha leído la letra "i".
#        – Solo se acepta "f": si se recibe "f", se transita a q3.
#        – Si se recibe cualquier otro carácter o un espacio (incompleto), se rechaza (qd).
#
#   q3: Se ha leído exactamente "if".
#        – Si se recibe un espacio, se interpreta como el final de la palabra, se emite TOKEN_IF y se retorna a q1.
#        – Cualquier otro carácter invalida la palabra: se transita a qd.
#
#   qd: Estado de rechazo (dead state). Aquí se permanece mientras se vayan ingresando letras
#       que no pueden formar "if". Al leer un espacio, se termina la palabra rechazada y se reinicia a q1.
#
# Nota: Se asume que al finalizar la entrada se simula la recepción de un espacio para que se valide la palabra.
#

# Definición del alfabeto: (mínimo se usan las letras necesarias y el espacio)
alphabet = list("abcdefghijklmnopqrstuvwxyz ")

# Estados definidos. Usamos q1 como estado "base" o de inicio en lugar de q0.
states = ["q1", "q2", "q3", "qd"]

# Estado inicial
initial_state = "q1"

# Transiciones y salidas:
transitions = {}
outputs = {}

# Desde el estado q1 (base):
for c in alphabet:
    if c == " ":
        # Un espacio en q1 mantiene la máquina en q1 (por si hay espacios consecutivos)
        transitions[("q1", c)] = "q1"
        outputs[("q1", c)] = ""
    elif c == "i":
        # Al recibir "i" se transita a q2
        transitions[("q1", c)] = "q2"
        outputs[("q1", c)] = ""
    else:
        # Cualquier otro carácter no forma parte de la palabra "if" y se rechaza
        transitions[("q1", c)] = "qd"
        outputs[("q1", c)] = ""

# Desde el estado q2 (se leyó "i"):
for c in alphabet:
    if c == "f":
        # Solo se acepta la letra "f": al recibirla se transita a q3
        transitions[("q2", c)] = "q3"
        outputs[("q2", c)] = ""
    elif c == " ":
        # Si se recibe un espacio después de "i", la palabra es incompleta y se rechaza
        transitions[("q2", c)] = "qd"
        outputs[("q2", c)] = ""
    else:
        # Cualquier otro carácter lleva al estado de rechazo
        transitions[("q2", c)] = "qd"
        outputs[("q2", c)] = ""

# Desde el estado q3 (se ha leído "if"):
for c in alphabet:
    if c == " ":
        # Al recibir un espacio se interpreta el fin de la palabra,
        # se emite TOKEN_IF y se retorna al estado base q1
        transitions[("q3", c)] = "q1"
        outputs[("q3", c)] = "TOKEN_IF"
    else:
        # Cualquier otro carácter invalida la palabra exacta "if"
        transitions[("q3", c)] = "qd"
        outputs[("q3", c)] = ""

# Desde el estado de rechazo (qd):
for c in alphabet:
    if c == " ":
        # Al terminar la palabra rechazada con un espacio, se reinicia a q1 sin token
        transitions[("qd", c)] = "q1"
        outputs[("qd", c)] = ""
    else:
        # Se permanece en el estado de rechazo mientras se sigan ingresando otros caracteres
        transitions[("qd", c)] = "qd"
        outputs[("qd", c)] = ""

# Esta máquina produce TOKEN_IF solo al recibir la secuencia:
# q1 --'i'--> q2 --'f'--> q3 --' '--> q1
# Es decir, la palabra "if" delimitada por espacios (o al comienzo/final de la
# entrada, suponiendo que se simula un espacio al final).
